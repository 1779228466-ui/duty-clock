# 问题1详尽解决方案：消除C++层全局变量

## 一、根因分析

当前两个JNI文件的**所有状态**都是文件级`static`全局变量：

| 文件 | 全局变量 | 后果 |
|------|----------|------|
| `neuron_direct_jni.cpp` | `g_state`(整个NPU状态)、`g_npu_mutex`、`g_neuron_lib`、14个`fn_xxx`函数指针 | 只能有一个NPU引擎实例；释放后若不手动清零所有字段则重建会崩溃 |
| `posemouse_ncnn_jni.cpp` | `g_net`、`g_use_gpu`、`g_target_size`、`g_kcp_*`(10个)、`g_abort_jmp`、`g_abort_expected` | 同上；KCP和NCNN耦合在同一编译单元；SIGABRT handler是全局的，多线程不安全 |

**核心问题**：JNI的`nativeDetect`等方法只接收Java参数，无法知道"哪个实例"在调用——因为状态是全局的，所以天然只能单实例。

## 二、设计原则

1. **实例化**：每个Java对象持有一个`jlong nativePtr`，指向C++堆上的Context对象
2. **RAII**：Context在`nativeCreate`中`new`，在`nativeDestroy`中`delete`，Kotlin层`release()`确保调用
3. **NeuronApi单例合理化**：系统SO库确实进程级只加载一次，保留为懒加载单例，但加线程安全
4. **KCP解耦**：从NCNN编译单元中拆出，每个`UdpFrameReceiver`持有独立KCP上下文
5. **SIGABRT handler线程安全化**：`sigjmp_buf`和`expected`标志改为`thread_local`

## 三、neuron_direct_jni.cpp 重构

### 3.2 NeuronDirectContext（实例化状态）

```cpp
struct PartitionInfo {
    NeuronCompilation* compilation = nullptr;
    NeuronExecution* execution = nullptr;
    int num_inputs = 0;
    int num_outputs = 0;
    std::vector<size_t> input_sizes;
    std::vector<size_t> output_sizes;
};

struct InputMapping {
    int src_partition;
    int src_output_idx;
};

static const InputMapping kDagMapping[3][10] = {
    {{-1, 0}},
    {{0, 5}, {0, 4}, {0, 3}, {0, 2}, {0, 1}, {0, 0}},
    {{1, 8}, {1, 7}, {1, 6}, {1, 4}, {1, 5}, {1, 1}, {1, 3}, {1, 2}, {1, 0}},
};

class NeuronDirectContext {
    std::vector<PartitionInfo> partitions_;
    std::vector<std::vector<uint8_t>> output_buffers_;
    std::vector<float> input_floats_;
    std::vector<int8_t> input_int8_;
    bool initialized_ = false;
    double last_infer_ms_ = 0;
    int log_counter_ = 0;
    bool use_int8_input_ = true;
    std::mutex mutex_;

    static constexpr size_t MAX_OUTPUTS_PER_PARTITION = 16;

    static inline float dequantizeInt8(int8_t val, float scale, int32_t zp) {
        return (static_cast<float>(val) - static_cast<float>(zp)) * scale;
    }

    bool initFromBytecodeDir(const char* bytecodeDir) {
        auto* api = NeuronApi::get();
        if (!api) {
            LOGE("Cannot load Neuron API");
            return false;
        }

        LOGI("=== INIT NPU DIRECT ENGINE === dir=%s", bytecodeDir);

        std::vector<std::string> bytecodePaths;
        std::vector<size_t> bytecodeSizes;

        // 尝试 Partition_X.bin 格式
        for (int i = 0; i < 20; i++) {
            char path[512];
            snprintf(path, sizeof(path), "%s/neuron_bytecode/Partition_%d.bin", bytecodeDir, i);
            FILE* f = fopen(path, "rb");
            if (!f) continue;
            fseek(f, 0, SEEK_END);
            long sz = ftell(f);
            fclose(f);
            if (sz > 0) {
                bytecodePaths.push_back(std::string(path));
                bytecodeSizes.push_back((size_t)sz);
            }
        }

        // 回退: neuron_bytecode_X.nb 格式
        if (bytecodePaths.empty()) {
            for (int i = 0; i < 20; i++) {
                char path[512];
                snprintf(path, sizeof(path), "%s/neuron_bytecode_%d.nb", bytecodeDir, i);
                FILE* f = fopen(path, "rb");
                if (!f) continue;
                fseek(f, 0, SEEK_END);
                long sz = ftell(f);
                fclose(f);
                if (sz > 0) {
                    bytecodePaths.push_back(std::string(path));
                    bytecodeSizes.push_back((size_t)sz);
                }
            }
        }

        if (bytecodePaths.empty()) {
            LOGE("No bytecode files found in %s", bytecodeDir);
            return false;
        }

        LOGI("Found %zu bytecode files", bytecodePaths.size());

        partitions_.clear();
        output_buffers_.clear();

        for (size_t i = 0; i < bytecodePaths.size(); i++) {
            PartitionInfo pi = {};

            FILE* f = fopen(bytecodePaths[i].c_str(), "rb");
            if (!f) {
                LOGE("Cannot open %s", bytecodePaths[i].c_str());
                return false;
            }

            std::vector<uint8_t> data(bytecodeSizes[i]);
            fread(data.data(), 1, bytecodeSizes[i], f);
            fclose(f);

            NeuronModel* model = nullptr;
            NeuronCompilation* compilation = nullptr;
            int result = api->modelRestore()(&model, &compilation, data.data(), bytecodeSizes[i]);
            if (result != NEURON_NO_ERROR) {
                LOGE("bytecode_%zu: model_restore failed (error=%d)", i, result);
                return false;
            }

            pi.compilation = compilation;

            for (int idx = 0; idx < 20; idx++) {
                size_t sz = 0;
                if (api->compilationGetInputPaddedSize()(compilation, idx, &sz) == NEURON_NO_ERROR && sz > 0) {
                    pi.num_inputs = idx + 1;
                    pi.input_sizes.push_back(sz);
                } else break;
            }

            for (int idx = 0; idx < 20; idx++) {
                size_t sz = 0;
                if (api->compilationGetOutputPaddedSize()(compilation, idx, &sz) == NEURON_NO_ERROR && sz > 0) {
                    pi.num_outputs = idx + 1;
                    pi.output_sizes.push_back(sz);
                } else break;
            }

            LOGI("  P%zu: %d inputs, %d outputs", i, pi.num_inputs, pi.num_outputs);

            // 维度日志
            for (int idx = 0; idx < pi.num_inputs; idx++) {
                uint32_t dims[8] = {};
                if (api->compilationGetInputPaddedDimensions()) {
                    api->compilationGetInputPaddedDimensions()(compilation, idx, dims);
                    LOGI("    P%zu in[%d]: size=%zu dims=[%u,%u,%u,%u,%u,%u,%u,%u]",
                         i, idx, pi.input_sizes[idx],
                         dims[0], dims[1], dims[2], dims[3],
                         dims[4], dims[5], dims[6], dims[7]);
                }
            }
            for (int idx = 0; idx < pi.num_outputs; idx++) {
                uint32_t dims[8] = {};
                if (api->compilationGetOutputPaddedDimensions()) {
                    api->compilationGetOutputPaddedDimensions()(compilation, idx, dims);
                    LOGI("    P%zu out[%d]: size=%zu dims=[%u,%u,%u,%u,%u,%u,%u,%u]",
                         i, idx, pi.output_sizes[idx],
                         dims[0], dims[1], dims[2], dims[3],
                         dims[4], dims[5], dims[6], dims[7]);
                }
            }

            NeuronExecution* execution = nullptr;
            result = api->executionCreate()(compilation, &execution);
            if (result != NEURON_NO_ERROR) {
                LOGE("bytecode_%zu: execution_create failed (error=%d)", i, result);
                api->compilationFree()(compilation);
                api->modelFree()(model);
                return false;
            }

            if (api->executionSetBoostHint()) {
                api->executionSetBoostHint()(execution, 100);
            }

            pi.execution = execution;
            partitions_.push_back(std::move(pi));
        }

        size_t maxOutputSize = 0;
        for (auto& pi : partitions_) {
            for (auto sz : pi.output_sizes) {
                if (sz > maxOutputSize) maxOutputSize = sz;
            }
        }

        output_buffers_.resize(partitions_.size() * MAX_OUTPUTS_PER_PARTITION);
        for (auto& buf : output_buffers_) {
            buf.resize(maxOutputSize + 4096, 0);
        }

        input_floats_.resize(1 * 320 * 320 * 3, 0);
        input_int8_.resize(1 * 320 * 320 * 3, 0);

        initialized_ = true;
        LOGI("=== NPU DIRECT ENGINE INIT OK: %zu partitions ===", partitions_.size());
        return true;
    }

    bool runInference(const float* inputData, int inputFloats,
                      float* outputData, int outputFloats) {
        if (!initialized_ || partitions_.empty()) return false;

        std::lock_guard<std::mutex> lock(mutex_);

        auto* api = NeuronApi::get();
        auto& parts = partitions_;
        int np = (int)parts.size();
        int npRun = np < 3 ? np : 3;

        // 诊断日志
        if (log_counter_ % 120 == 0) {
            float minInp = 999, maxInp = -999;
            for (int k = 0; k < std::min(inputFloats, 1000); k++) {
                if (inputData[k] < minInp) minInp = inputData[k];
                if (inputData[k] > maxInp) maxInp = inputData[k];
            }
            LOGI("INPUT range: [%.4f,%.4f]", minInp, maxInp);
        }

        auto t0 = std::chrono::high_resolution_clock::now();

        for (int p = 0; p < npRun; p++) {
            auto& pi = parts[p];
            int result = NEURON_NO_ERROR;
            NeuronExecution* exec = pi.execution;

            for (int idx = 0; idx < pi.num_inputs; idx++) {
                size_t dataSize = pi.input_sizes[idx];
                uint8_t* dataPtr = nullptr;

                if (p == 0 && idx == 0) {
                    if (use_int8_input_) {
                        for (int k = 0; k < inputFloats; k++) {
                            input_int8_[k] = (int8_t)(inputData[k] * 255.0f - 128.0f);
                        }
                        dataPtr = (uint8_t*)input_int8_.data();
                        dataSize = inputFloats;
                    } else {
                        dataPtr = (uint8_t*)inputData;
                        dataSize = inputFloats * sizeof(float);
                    }
                } else {
                    auto& mapping = kDagMapping[p][idx];
                    int sp = mapping.src_partition;
                    int soi = mapping.src_output_idx;
                    if (sp >= 0 && sp < np) {
                        dataPtr = output_buffers_[sp * MAX_OUTPUTS_PER_PARTITION + soi].data();
                        dataSize = parts[sp].output_sizes[soi];
                    }
                    if (!dataPtr) {
                        LOGW("P%d input[%d]: mapping invalid, using zero buf", p, idx);
                        static thread_local std::vector<uint8_t> zeroBuf(4 * 1024 * 1024, 0);
                        dataPtr = zeroBuf.data();
                    }
                }

                if (log_counter_ <= 6 && p <= 2) {
                    int8_t* di = (int8_t*)dataPtr;
                    LOGI("P%d.in[%d] size=%zu first8i8=[%d,%d,%d,%d,%d,%d,%d,%d]",
                         p, idx, dataSize,
                         di[0], di[1], di[2], di[3], di[4], di[5], di[6], di[7]);
                }

                result = api->executionSetInput()(exec, idx, nullptr, dataPtr, dataSize);
                if (result != NEURON_NO_ERROR) {
                    LOGE("P%d setInput[%d] failed (error=%d)", p, idx, result);
                    return false;
                }
            }

            for (int idx = 0; idx < pi.num_outputs; idx++) {
                size_t sz = pi.output_sizes[idx];
                if (output_buffers_[p * MAX_OUTPUTS_PER_PARTITION + idx].size() < sz) {
                    output_buffers_[p * MAX_OUTPUTS_PER_PARTITION + idx].resize(sz);
                }
                result = api->executionSetOutput()(exec, idx, nullptr,
                    output_buffers_[p * MAX_OUTPUTS_PER_PARTITION + idx].data(), sz);
                if (result != NEURON_NO_ERROR) {
                    LOGE("P%d setOutput[%d] failed (error=%d)", p, idx, result);
                    return false;
                }
            }

            result = api->executionCompute()(exec);
            if (result != NEURON_NO_ERROR) {
                LOGE("P%d compute failed (error=%d)", p, result);
                return false;
            }

            // 输出诊断日志
            if (log_counter_ <= 6 || log_counter_ % 120 == 0) {
                for (int oi = 0; oi < pi.num_outputs; oi++) {
                    int8_t* buf = (int8_t*)output_buffers_[p * MAX_OUTPUTS_PER_PARTITION + oi].data();
                    size_t sz = pi.output_sizes[oi];
                    int8_t mn = 127, mx = -128;
                    int n = (int)std::min(sz, (size_t)4096);
                    for (int k = 0; k < n; k++) {
                        if (buf[k] < mn) mn = buf[k];
                        if (buf[k] > mx) mx = buf[k];
                    }
                    float* fp = (float*)buf;
                    float fmn = 1e30, fmx = -1e30;
                    int nf = (int)std::min(sz / 4, (size_t)1024);
                    for (int k = 0; k < nf; k++) {
                        if (fp[k] < fmn) fmn = fp[k];
                        if (fp[k] > fmx) fmx = fp[k];
                    }
                    LOGI("P%d.out[%d] INT8=[%d,%d] FP32=[%f,%f] first8i8=[%d,%d,%d,%d,%d,%d,%d,%d]",
                         p, oi, mn, mx, fmn, fmx,
                         buf[0], buf[1], buf[2], buf[3], buf[4], buf[5], buf[6], buf[7]);
                }
            }
        }

        auto t1 = std::chrono::high_resolution_clock::now();
        double npuMs = std::chrono::duration<double, std::milli>(t1 - t0).count();

        // P2后处理
        static constexpr float kScale = 0.004445722792297602f;
        static constexpr int32_t kZp = -115;
        static constexpr int kNumAnchors = 2100;
        static constexpr int kNumFields = 56;
        static constexpr int kTopK = 300;

        int8_t* rawPreds = (int8_t*)output_buffers_[2 * MAX_OUTPUTS_PER_PARTITION + 0].data();
        int8_t* rawConf = (int8_t*)output_buffers_[2 * MAX_OUTPUTS_PER_PARTITION + 2].data();

        // 诊断日志
        if (log_counter_ % 60 == 0) {
            int8_t maxI8 = -128, minI8 = 127;
            for (int k = 0; k < kNumAnchors; k++) {
                if (rawConf[k] > maxI8) maxI8 = rawConf[k];
                if (rawConf[k] < minI8) minI8 = rawConf[k];
            }
            LOGI("P2.out[2] INT8 range: [%d,%d] dequant range: [%.4f,%.4f]",
                 minI8, maxI8,
                 (minI8 - kZp) * kScale, (maxI8 - kZp) * kScale);

            int8_t maxPred = -128, minPred = 127;
            for (int k = 0; k < std::min((size_t)1000, parts[2].output_sizes[0]); k++) {
                if (rawPreds[k] > maxPred) maxPred = rawPreds[k];
                if (rawPreds[k] < minPred) minPred = rawPreds[k];
            }
            LOGI("P2.out[0] INT8 range: [%d,%d] dequant range: [%.4f,%.4f]",
                 minPred, maxPred,
                 (minPred - kZp) * kScale, (maxPred - kZp) * kScale);

            LOGI("P2.out[2] first20: [%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d]",
                 rawConf[0], rawConf[1], rawConf[2], rawConf[3], rawConf[4],
                 rawConf[5], rawConf[6], rawConf[7], rawConf[8], rawConf[9],
                 rawConf[10], rawConf[11], rawConf[12], rawConf[13], rawConf[14],
                 rawConf[15], rawConf[16], rawConf[17], rawConf[18], rawConf[19]);

            LOGI("P2.out[0] anchor0: [%d,%d,%d,%d, %d, %d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d,%d]",
                 rawPreds[0], rawPreds[1], rawPreds[2], rawPreds[3],
                 rawPreds[4],
                 rawPreds[5], rawPreds[6], rawPreds[7],
                 rawPreds[8], rawPreds[9], rawPreds[10],
                 rawPreds[11], rawPreds[12], rawPreds[13],
                 rawPreds[14], rawPreds[15], rawPreds[16],
                 rawPreds[17], rawPreds[18], rawPreds[19]);

            int8_t* p2out1 = (int8_t*)output_buffers_[2 * MAX_OUTPUTS_PER_PARTITION + 1].data();
            LOGI("P2.out[1] first8: [%d,%d,%d,%d,%d,%d,%d,%d]",
                 p2out1[0], p2out1[1], p2out1[2], p2out1[3],
                 p2out1[4], p2out1[5], p2out1[6], p2out1[7]);

            int8_t* p2out3 = (int8_t*)output_buffers_[2 * MAX_OUTPUTS_PER_PARTITION + 3].data();
            LOGI("P2.out[3] first8 INT8: [%d,%d,%d,%d,%d,%d,%d,%d]",
                 p2out3[0], p2out3[1], p2out3[2], p2out3[3],
                 p2out3[4], p2out3[5], p2out3[6], p2out3[7]);
        }

        std::vector<std::pair<float, int>> scores(kNumAnchors);
        for (int i = 0; i < kNumAnchors; i++) {
            float logit = dequantizeInt8(rawConf[i], kScale, kZp);
            float conf = 1.0f / (1.0f + expf(-logit));
            scores[i] = {conf, i};
        }

        std::partial_sort(scores.begin(), scores.begin() + kTopK, scores.end(),
                          [](const auto& a, const auto& b) { return a.first > b.first; });

        memset(outputData, 0, kTopK * 57 * sizeof(float));

        for (int i = 0; i < kTopK; i++) {
            int anchor = scores[i].second;
            float conf = scores[i].first;
            int baseOut = i * 57;

            outputData[baseOut + 4] = conf;

            for (int j = 0; j < 4; j++) {
                int8_t val = rawPreds[anchor * kNumFields + j];
                float fv = dequantizeInt8(val, kScale, kZp);
                outputData[baseOut + j] = tanhf(fv);
            }

            for (int k = 0; k < 17; k++) {
                for (int j = 0; j < 3; j++) {
                    int8_t val = rawPreds[anchor * kNumFields + 5 + k * 3 + j];
                    float fv = dequantizeInt8(val, kScale, kZp);
                    outputData[baseOut + 5 + k * 3 + j] = tanhf(fv);
                }
            }
        }

        auto t2 = std::chrono::high_resolution_clock::now();
        double totalMs = std::chrono::duration<double, std::milli>(t2 - t0).count();
        last_infer_ms_ = totalMs;

        if (log_counter_ % 30 == 0) {
            float maxConf = 0;
            for (int i = 0; i < kTopK; i++) {
                if (scores[i].first > maxConf) maxConf = scores[i].first;
            }
            LOGI("npu=%.1fms post=%.1fms total=%.1fms maxConf=%.4f top5=[%.4f,%.4f,%.4f,%.4f,%.4f]",
                 npuMs, totalMs - npuMs, totalMs, maxConf,
                 scores[0].first, scores[1].first, scores[2].first, scores[3].first, scores[4].first);
        }
```

```cpp
class OmpInitGuard {
    static thread_local sigjmp_buf jmpBuf_;
    static thread_local volatile sig_atomic_t expected_;
    struct sigaction oldSa_;

    static void handler(int sig) {
        if (expected_) siglongjmp(jmpBuf_, 1);
    }

public:
    OmpInitGuard() {
        expected_ = 1;
        struct sigaction sa;
        sa.sa_handler = handler;
        sa.sa_flags = 0;
        sigemptyset(&sa.sa_mask);
        sigaction(SIGABRT, &sa, &oldSa_);
    }

    ~OmpInitGuard() {
        sigaction(SIGABRT, &oldSa_, nullptr);
        expected_ = 0;
    }

    sigjmp_buf& jmpBuf() { return jmpBuf_; }

    static void suppressAffinity() {
        setenv("KMP_AFFINITY", "disabled", 1);
        setenv("OMP_PROC_BIND", "false", 1);
        setenv("OMP_NUM_THREADS", "1", 1);
        setenv("KMP_SETTINGS", "0", 1);
        omp_set_num_threads(1);
        omp_set_dynamic(0);
    }
};
```

### 4.3 KcpContext（实例化，从NCNN编译单元解耦）

```cpp
class KcpContext {
    int sock_ = -1;
    struct sockaddr_in destAddr_;
    ikcpcb* kcp_ = nullptr;
    volatile bool running_ = false;
    pthread_t thread_ = 0;

    uint8_t* frameBuf_ = nullptr;
    int frameBufSize_ = 0;
    volatile int frameWidth_ = 0;
    volatile int frameLen_ = 0;
    volatile int frameId_ = -1;
    volatile int frameReady_ = 0;
    pthread_mutex_t frameMutex_ = PTHREAD_MUTEX_INITIALIZER;
    uint64_t recvCount_ = 0;
    uint64_t recvBytes_ = 0;

    static int kcpUdpSend(const char* buf, int len, struct IKCPCB* kcp, void* user) {
        auto* ctx = static_cast<KcpContext*>(user);
        if (ctx->sock_ < 0) return -1;
        int ret = sendto(ctx->sock_, buf, len, 0,
                         (struct sockaddr*)&ctx->destAddr_, sizeof(ctx->destAddr_));
        return (ret < 0) ? -1 : 0;
    }

    static void* recvThreadFunc(void* arg) {
        auto* ctx = static_cast<KcpContext*>(arg);
        OmpInitGuard::suppressAffinity();

        uint8_t recvBuf[65536];
        struct sockaddr_in fromAddr;
        socklen_t fromLen = sizeof(fromAddr);

        LOGI("KCP recv thread started");

        while (ctx->running_) {
            int batchCount = 0;
            while (ctx->running_) {
                socklen_t fl = sizeof(fromAddr);
                ssize_t n = recvfrom(ctx->sock_, recvBuf, sizeof(recvBuf), MSG_DONTWAIT,
                                     (struct sockaddr*)&fromAddr, &fl);
                if (n <= 0) break;

                if (ctx->kcp_) {
                    ikcp_input(ctx->kcp_, (const char*)recvBuf, (int)n);
                    ctx->recvCount_++;
                    ctx->recvBytes_ += n;
                    batchCount++;

                    if (ctx->recvCount_ == 1) {
                        memcpy(&ctx->destAddr_, &fromAddr, sizeof(fromAddr));
                        char addrStr[64];
                        inet_ntop(AF_INET, &fromAddr.sin_addr, addrStr, sizeof(addrStr));
                        LOGI("KCP learned peer: %s:%d", addrStr, ntohs(fromAddr.sin_port));
                    }
                }
            }

            if (ctx->kcp_) {
                uint32_t currentMs = (uint32_t)(std::chrono::duration_cast<std::chrono::milliseconds>(
                    std::chrono::steady_clock::now().time_since_epoch()).count());
                ikcp_update(ctx->kcp_, currentMs);

                while (true) {
                    char kcpBuf[65536];
                    int kcpLen = ikcp_recv(ctx->kcp_, kcpBuf, sizeof(kcpBuf));
                    if (kcpLen <= 0) break;

                    if (kcpLen >= 16) {
                        int dataLen = kcpLen - 16;
                        if (dataLen > 0) {
                            pthread_mutex_lock(&ctx->frameMutex_);
                            int needed = dataLen + 16;
                            if (ctx->frameBufSize_ < needed) {
                                delete[] ctx->frameBuf_;
                                ctx->frameBuf_ = new uint8_t[needed + 4096];
                                ctx->frameBufSize_ = needed + 4096;
                            }
                            memcpy(ctx->frameBuf_, kcpBuf, 16);
                            memcpy(ctx->frameBuf_ + 16, kcpBuf + 16, dataLen);
                            ctx->frameLen_ = dataLen + 16;

                            uint32_t fi32;
                            memcpy(&fi32, kcpBuf, 4);
                            ctx->frameWidth_ = (int)fi32;
                            ctx->frameId_ = (int)fi32;
                            ctx->frameReady_ = 1;
                            pthread_mutex_unlock(&ctx->frameMutex_);
                        }

                        static int kcpRecvFrameLog = 0;
                        kcpRecvFrameLog++;
                        if (kcpRecvFrameLog <= 10 || kcpRecvFrameLog % 100 == 0) {
                            uint32_t fi32, rawSize, compSize;
                            uint16_t pktIdx, totalPkts;
                            memcpy(&fi32, kcpBuf, 4);
                            memcpy(&pktIdx, kcpBuf + 4, 2);
                            memcpy(&totalPkts, kcpBuf + 6, 2);
                            memcpy(&rawSize, kcpBuf + 8, 4);
                            memcpy(&compSize, kcpBuf + 12, 4);
                            LOGI("KCP recv frame #%d: fid=%u comp=%u raw=%u data_len=%d kcp_len=%d",
                                 kcpRecvFrameLog, fi32, compSize, rawSize, dataLen, kcpLen);
                        }
                    }
                }

                if (ctx->recvCount_ > 0 && ctx->recvCount_ % 500 == 0) {
                    LOGI("KCP stats: input=%llu bytes=%llu nrcv_buf=%u nsnd_buf=%u nrcv_que=%u nsnd_que=%u",
                         (unsigned long long)ctx->recvCount_, (unsigned long long)ctx->recvBytes_,
                         ctx->kcp_->nrcv_buf, ctx->kcp_->nsnd_buf, ctx->kcp_->nrcv_que, ctx->kcp_->nsnd_que);
                }
            }

            if (batchCount == 0) {
                usleep(500);
            }
        }

        LOGI("KCP recv thread stopped");
        return nullptr;
    }

public:
    static KcpContext* fromPtr(jlong ptr) {
        return reinterpret_cast<KcpContext*>(ptr);
    }
    jlong toPtr() const {
        return reinterpret_cast<jlong>(const_cast<KcpContext*>(this));
    }

    bool start(int port, int conv) {
        if (running_) stop();

        sock_ = socket(AF_INET, SOCK_DGRAM, 0);
        if (sock_ < 0) {
            LOGE("KCP socket create failed: %s", strerror(errno));
            return false;
        }

        int rcvbuf = 8 * 1024 * 1024;
        setsockopt(sock_, SOL_SOCKET, SO_RCVBUF, &rcvbuf, sizeof(rcvbuf));

        struct timeval tv;
        tv.tv_sec = 0; tv.tv_usec = 500000;
        setsockopt(sock_, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));

        struct sockaddr_in localAddr;
        memset(&localAddr, 0, sizeof(localAddr));
        localAddr.sin_family = AF_INET;
        localAddr.sin_port = htons((uint16_t)port);
        localAddr.sin_addr.s_addr = INADDR_ANY;

        if (bind(sock_, (struct sockaddr*)&localAddr, sizeof(localAddr)) < 0) {
            LOGE("KCP bind failed: %s", strerror(errno));
            close(sock_); sock_ = -1;
            return false;
        }

        memset(&destAddr_, 0, sizeof(destAddr_));
        destAddr_.sin_family = AF_INET;
        destAddr_.sin_port = htons((uint16_t)port);
        destAddr_.sin_addr.s_addr = htonl(INADDR_ANY);

        // 排空残留包
        {
            uint8_t drainBuf[65536];
            struct sockaddr_in drainAddr;
            socklen_t drainLen = sizeof(drainAddr);
            int drainCount = 0;
            while (true) {
                ssize_t dn = recvfrom(sock_, drainBuf, sizeof(drainBuf), MSG_DONTWAIT,
                                       (struct sockaddr*)&drainAddr, &drainLen);
                if (dn <= 0) break;
                drainCount++;
            }
            LOGI("KCP drained %d stale packets from socket", drainCount);
        }

        kcp_ = ikcp_create((IUINT32)conv, this);  // user指针传this
        if (kcp_) {
            ikcp_nodelay(kcp_, 1, 1, 2, 1);
            ikcp_setmtu(kcp_, 1400);
            ikcp_wndsize(kcp_, 4096, 4096);
            kcp_->stream = 0;
            kcp_->output = kcpUdpSend;
        }

        recvCount_ = 0;
        recvBytes_ = 0;
        frameReady_ = 0;
        running_ = true;

        pthread_create(&thread_, nullptr, recvThreadFunc, this);  // 传this
        LOGI("KCP started: port=%d conv=0x%08X", port, conv);
        return true;
    }

    bool getFrame(uint8_t* outBuf, int outBufSize,
                  int* outFrameId, int* outRawSize, int* outCompSize,
                  int* outDataLen, int* outTotalPkts) {
        pthread_mutex_lock(&frameMutex_);
        if (!frameReady_ || frameLen_ <= 16) {
            pthread_mutex_unlock(&frameMutex_);
            return false;
        }

        if (outBuf && outBufSize >= frameLen_) {
            memcpy(outBuf, frameBuf_, frameLen_);
        }

        uint32_t fi32, rawSize, compSize;
        uint16_t pktIdx, totalPkts;
        memcpy(&fi32, frameBuf_, 4);
        memcpy(&pktIdx, frameBuf_ + 4, 2);
        memcpy(&totalPkts, frameBuf_ + 6, 2);
        memcpy(&rawSize, frameBuf_ + 8, 4);
        memcpy(&compSize, frameBuf_ + 12, 4);

        *outFrameId = (int)fi32;
        *outRawSize = (int)rawSize;
        *outCompSize = (int)compSize;
        *outDataLen = frameLen_ - 16;
        *outTotalPkts = (int)totalPkts;

        frameReady_ = 0;
        pthread_mutex_unlock(&frameMutex_);
        return true;
    }

    void stop() {
        running_ = false;
        if (thread_) {
            pthread_join(thread_, nullptr);
            thread_ = 0;
        }
        if (kcp_) {
            ikcp_release(kcp_);
            kcp_ = nullptr;
        }
        if (sock_ >= 0) {
            close(sock_);
            sock_ = -1;
        }
        if (frameBuf_) {
            delete[] frameBuf_;
            frameBuf_ = nullptr;
        }
        frameBufSize_ = 0;
        frameReady_ = 0;
        frameId_ = -1;
        frameWidth_ = 0;
        frameLen_ = 0;
        recvCount_ = 0;
        recvBytes_ = 0;
    }

    ~KcpContext() {
        stop();
    }
};
```

```kotlin
            } catch (e: Exception) {

            }
            return true

            AppLog.e(TAG, "loadModel failed", e)
            return false
        }

        if (!loaded || nativePtr == 0L) return emptyList()
        try {
            val w = bitmap.width; val h = bitmap.height
            val needResize = w != 320 || h != 320
            val src = if (needResize) Bitmap.createScaledBitmap(bitmap, 320, 320, true) else bitmap

            val t0 = System.nanoTime()
            src.getPixels(pixels, 0, 320, 0, 0, 320, 320)
            val t1 = System.nanoTime()

            val result = nativeDetect(nativePtr, pixels, 320, 320, confThresh)
            val t2 = System.nanoTime()
            lastInferMsVal = (t2 - t1) / 1e6

            if (needResize && src !== bitmap) src.recycle()

            if (result == null || result.isEmpty()) return emptyList()

            val numDets = result.size / 57
            val results = mutableListOf<PoseResult>()
            for (i in 0 until numDets) {
                val base = i * 57
                val conf = result[base + 4]
                if (conf < confThresh) continue
                val kps = mutableListOf<Keypoint>()

                    }
                }
                results.add(PoseResult(conf, kps))
            }
            return results
        } catch (e: Exception) {
            AppLog.e(TAG, "detect failed", e)
            return emptyList()
        }
    }
```

```kotlin
class UdpFrameReceiver(private val port: Int = 19876) {
    // ... 其他字段不变 ...

    private var kcpNativePtr: Long = 0

    fun setKcpMode(conv: Int = 0x12345678) {
        useKcp = true
        kcpConv = conv
    }

    private fun startKcpMode() {
        try {
            try { System.loadLibrary("posemouse_ncnn") } catch (_: UnsatisfiedLinkError) {}
            kcpNativePtr = nativeKcpCreate(port, kcpConv)
            if (kcpNativePtr == 0L) {
                AppLog.e(TAG, "KCP create failed")
                return
            }
            isRunning = true
            h264Decoder = H264Decoder()
            AppLog.i(TAG, "KCP模式启动 port=$port conv=0x${kcpConv.toString(16)}")

            Thread({
                while (!stopped) {
                    try {
                        val info = nativeKcpGetFrame(kcpNativePtr, h264Buf)
                        if (info != null && info.size == 5) {
                            val fid = info[0]
                            val rawSize = info[1]
                            val compSize = info[2]
                            val dataLen = info[3]
                            val totalPkts = info[4]

                            if (dataLen > 0 && compSize > 0) {
                                val w = calcWidth(rawSize)
                                if (w > 0) {
                                    val decoder = h264Decoder
                                    if (decoder != null) {
                                        decoder.configure(w, w)
                                        val bgra = decoder.decodeFrame(h264Buf, 16, dataLen)
                                        if (bgra != null && bgra.size == rawSize) {
                                            synchronized(lock) {
                                                System.arraycopy(bgra, 0, frameBuf, 0, rawSize)
                                                latestFrameId = fid
                                                latestWidth = w
                                                latestHeight = w
                                            }
                                            frameCount++
                                        }
                                    }
                                }
                            }
                        } else {
                            Thread.sleep(1)
                        }
                    } catch (e: Exception) {
                        if (!stopped) AppLog.w(TAG, "KCP getFrame error: ${e.message}")
                    }
                }
                isRunning = false
            }, "KcpRecv").start()
        } catch (e: Exception) {
            AppLog.e(TAG, "KCP启动失败", e)
        }
    }

    fun stop() {
        stopped = true
        h264Decoder?.release()
        h264Decoder = null
        if (useKcp && kcpNativePtr != 0L) {
            nativeKcpDestroy(kcpNativePtr)
            kcpNativePtr = 0
        }
        isRunning = false
    }

    // JNI签名变更
    private external fun nativeKcpCreate(port: Int, conv: Int): Long
    private external fun nativeKcpGetFrame(ptr: Long, outBuf: ByteArray): IntArray?
    private external fun nativeKcpDestroy(ptr: Long)

    companion object { private const val TAG = "UdpFrameReceiver" }
}
```
